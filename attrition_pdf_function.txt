
  async function exportAttritionHistoryPDF() {
    if (!canManage) return

    setHistoryError('')

    const employeeFilter = historyEmployeeFilter.trim()
    const riskFilter = historyRiskFilter
    const startDateFilter = historyStartDateFilter
    const endDateFilter = historyEndDateFilter
    const modelTypeFilter = historyModelTypeFilter.trim()

    if (
      startDateFilter &&
      endDateFilter &&
      startDateFilter > endDateFilter
    ) {
      setHistoryError('Start date must be on or before end date.')
      return
    }

    const params = {}

    if (employeeFilter) {
      const employeeId = Number(employeeFilter)

      if (!Number.isInteger(employeeId) || employeeId <= 0) {
        setHistoryError('Employee filter must be a positive integer ID.')
        return
      }

      params.employee_id = employeeId
    }

    if (riskFilter) params.risk_level = riskFilter
    if (startDateFilter) params.start_date = startDateFilter
    if (endDateFilter) params.end_date = endDateFilter
    if (modelTypeFilter) params.model_type = modelTypeFilter

    const filterSummary = [
      employeeFilter ? `Employee ID: ${employeeFilter}` : '',
      riskFilter ? `Risk Level: ${riskFilter}` : '',
      startDateFilter ? `Start Date: ${startDateFilter}` : '',
      endDateFilter ? `End Date: ${endDateFilter}` : '',
      modelTypeFilter ? `Model Type: ${modelTypeFilter}` : '',
    ].filter(Boolean)

    const printWindow = window.open('', '_blank')

    if (!printWindow) {
      setHistoryError('Allow pop-ups in your browser to generate the PDF report.')
      return
    }

    printWindow.document.open()
    printWindow.document.write(
      '<!DOCTYPE html><html><head><title>Attrition History Report</title></head>' +
      '<body style="font-family:Arial,sans-serif;padding:24px">' +
      '<h2>WorkForce AI Pro â€” Attrition History</h2>' +
      '<p>Loading prediction history...</p></body></html>'

      const pageSize = 100
      let page = 1
      let total = 0
      const allRecords = []
      const maxPages = 10000

      while (page <= maxPages) {
        const result = await attritionAPI.history({
          ...params,
          page,
          page_size: pageSize,
        })

        const items = Array.isArray(result.items) ? result.items : []
        total = Number(result.total) || 0

        if (items.length === 0) break

        allRecords.push(...items)

        if (allRecords.length >= total || items.length < pageSize) break

        page += 1
      }

      if (page > maxPages) {
        throw new Error('PDF export stopped because the pagination limit was reached.')
      }

      const escapeHTML = (value) =>
        String(value ?? '').replace(/[&<>"']/g, (character) => ({
          '&': '&amp;',
          '<': '&lt;',
          '>': '&gt;',
          '"': '&quot;',
          "'": '&#39;',
        })[character])

      const reportDate = new Date().toLocaleString()

      const rowsHTML = allRecords.map((record) => {
        const date = record.created_at
          ? new Date(record.created_at).toLocaleString()
          : ''

        const employee = record.employee_id == null
          ? 'General prediction'
          : record.employee_id

        const probability = record.attrition_risk_percentage ??
          (record.attrition_probability == null
            ? ''
            : Number(record.attrition_probability) * 100)

        const predictedBy = record.predicted_by_name ||
          (record.predicted_by_user_id != null
            ? `User #${record.predicted_by_user_id}`
            : '')

        return `<tr>
          <td>${escapeHTML(date)}</td>
          <td>${escapeHTML(employee)}</td>
          <td>${escapeHTML(record.risk_level)}</td>
          <td>${escapeHTML(probability === '' ? '' : Number(probability).toFixed(2) + '%')}</td>

          <td>${escapeHTML(predictedBy)}</td>
        </tr>`
      }).join('')
